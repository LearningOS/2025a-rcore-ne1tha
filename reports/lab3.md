### 题3

简易比较

```rust
	fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        const BIG_STRIDE: u64 = 255; // 以8位为例
        
        let a = self.0;
        let b = other.0;
        

        if a == b {
            Some(Ordering::Equal)
        } else if a.wrapping_sub(b) <= BIG_STRIDE / 2 {
            Some(b.cmp(&a))
        } else {
            // 这里溢出
            Some(Ordering::Greater)
        }
    }
```
